Privilege Management Script (privileges.sql) 
-- 1. Create Dedicated Application User
CREATE USER 'college_app_user'@'localhost' IDENTIFIED BY 'AppUser@Pass2026';
GRANT SELECT, INSERT, UPDATE, DELETE, EXECUTE, SHOW VIEW 
ON college_management.* TO 'college_app_user'@'localhost';

-- 2. Create Administrator Account
CREATE USER 'college_admin'@'localhost' IDENTIFIED BY 'Admin@Pass2026';
GRANT ALL PRIVILEGES ON college_management.* TO 'college_admin'@'localhost' WITH GRANT OPTION;

-- 3. Create Faculty User
CREATE USER 'college_faculty'@'localhost' IDENTIFIED BY 'Faculty@Pass2026';
GRANT SELECT ON college_management.* TO 'college_faculty'@'localhost';
GRANT INSERT, UPDATE ON college_management.attendance TO 'college_faculty'@'localhost';
GRANT INSERT, UPDATE ON college_management.results TO 'college_faculty'@'localhost';

-- 4. Create Clerk User and Demonstrate REVOKE
CREATE USER 'college_clerk'@'localhost' IDENTIFIED BY 'Clerk@Pass2026';
GRANT SELECT, INSERT, UPDATE, DELETE ON college_management.fees TO 'college_clerk'@'localhost';
REVOKE DELETE ON college_management.fees FROM 'college_clerk'@'localhost'; -- REVOKE Demonstration!

-- 5. Create Read-Only User
CREATE USER 'college_reader'@'localhost' IDENTIFIED BY 'Reader@Pass2026';
GRANT SELECT, SHOW VIEW ON college_management.* TO 'college_reader'@'localhost';

FLUSH PRIVILEGES;
